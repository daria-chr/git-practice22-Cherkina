git version — показывает установленную версию Git
git config --global user.name "Ник" — задаёт имя для коммитов
git config --global user.email "@почта." — задаёт email для коммитов
cd папка — переход в указанную папку
mkdir название — создаёт новую папку
git init — создаёт локальный Git-репозиторий в текущей папке
git remote add origin ссылка — привязывает локальный репозиторий к удалённому
git remote -v — проверяет, к каким удалённым репозиториям подключён проект
dotnet new console -o Program --use-program-main — создаёт консольное приложение C# с классической структурой (с методом Main)
git status — показывает, какие файлы изменены и готовы к коммиту
git add . — добавляет ВСЕ изменения в индекс
git commit -m "Сообщение" — фиксирует изменения с комментарием
git log — выводит полную историю коммитов
git log --oneline — компактный вывод истории 
git show хеш_коммита — показывает детальную информацию о конкретном коммите
git checkout -b имя_ветки — создаёт новую ветку и сразу переключается на неё
git branch — список всех локальных веток
git switch -c имя_ветки — создать ветку и переключиться
git switch имя_ветки — переключиться на существующую ветку
git pull — забирает изменения с удалённого репозитория и сливает с текущей веткой
git fetch --prune — состояние сервера

примеры есть на скринах работы.
